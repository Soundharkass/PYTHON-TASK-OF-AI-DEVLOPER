#7.Recursive function to find LCM.
'''
def gcd(a, b):
    if b == 0:
        return a
    else:
        return gcd(b, a % b)

def lcm(a, b):
    return (a * b) // gcd(a, b)

x = int(input("Enter first number: "))
y = int(input("Enter second number: "))

print("LCM of", x, "and", y, "is:", lcm(x, y))

def is_palindrome_number(n, s=None):
    if s is None:
        s = str(n)
    if len(s) <= 1:
        return True
    if s[0] != s[-1]:
        return False

    return is_palindrome_number(n, s[1:-1])

num = int(input("Enter a number: "))

#14.Recursive function to find nth Fibonacci number
'''
def fibonacci(n):
    if n <= 1:  
        return n
    else:        
        return fibonacci(n-1) + fibonacci(n-2)
def fibonacci_series(n):
    for i in range(n):
        print(fibonacci(i), end=" ")

num = int(input("Enter number of terms: "))
print("Fibonacci Series:")
fibonacci_series(num)

#15.Recursive function to count digits in number.
'''
def count_digits(n):
    # Base case: single digit number
    if n < 10:
        return 1
    else:
        # Recursive case: remove last digit and add 1
        return 1 + count_digits(n // 10)

# Example usage
num = int(input("Enter a number: "))
print("Number of digits:", count_digits(num))

#16.Recursive function to find maximum element in list
'''
def find_max(lst, n=None):
    if n is None:
        n = len(lst) - 1
    
    if n == 0:
        return lst[0]
    
    return max(lst[n], find_max(lst, n-1))

numbers = [12, 45, 7, 89, 23, 56]
print("Maximum element:", find_max(numbers))

#18.Recursive function to remove all spaces in a string
'''
def remove_spaces(s):
    if s == "":
        return ""
    if s[0] == " ":
        return remove_spaces(s[1:])

    return s[0] + remove_spaces(s[1:])

text = "Hello World Program in Python"
print("Original String:", text)
print("String without spaces:", remove_spaces(text))

def count_vowels(s):
    vowels = "aeiouAEIOU"
    
    if s == "":
        return 0
    
    if s[0] in vowels:
        return 1 + count_vowels(s[1:])
    else:
        return count_vowels(s[1:])

#20.Recursive function to check prime number
'''
def is_prime(n, i=2):
    if n <= 1:
        return False
    if i * i > n:   
        return True
    if n % i == 0:  
        return False
    
    return is_prime(n, i+1)

num = int(input("Enter a number: "))
if is_prime(num):
    print(num, "is a Prime Number")
else:
    print(num, "is Not a Prime Number")
